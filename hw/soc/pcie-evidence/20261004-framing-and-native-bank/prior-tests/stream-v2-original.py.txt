ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "scripts"))
import characterize_pcie_clock_trim_stream_v2 as m  # noqa: E402


def await_file(path, process, timeout=4):
    deadline = time.monotonic() + timeout
    while not path.exists():
        assert process.poll() is None, (process.returncode, path)
        assert time.monotonic() < deadline, path
        time.sleep(0.01)
    return json.loads(path.read_text())


def stopped(identity):
    observations = []
    deadline = time.monotonic() + 1
    while True:
        actual = m.process_identity(identity["pid"])
        observations.append(dict(monotonic=time.monotonic(), identity=actual))
        gone = actual is None or actual["start_ticks"] != identity["start_ticks"]
        if gone or actual["state"] == "Z":
            return observations
        assert time.monotonic() < deadline, observations
        time.sleep(0.005)


def publisher_text(mode):
    return """import json,os,pathlib,signal,sys,time
root=pathlib.Path(sys.argv[sys.argv.index('--out')+1]).parent
mode=MODE
if mode=='publisher_nonzero':raise SystemExit(17)
if mode=='interrupt_descendant' or mode=='exited_leader':
 child=os.fork()
 if child:
  while not (root/'publisher-armed.json').exists():time.sleep(.002)
  if mode=='exited_leader':raise SystemExit(0)
  while True:time.sleep(.1)
signal.signal(signal.SIGTERM,signal.SIG_IGN)
fields=pathlib.Path('/proc',str(os.getpid()),'stat').read_text().rsplit(') ',1)[1].split()
deadline=time.monotonic()+7.0
(root/'publisher-armed.json').write_text(json.dumps(dict(pid=os.getpid(),start_ticks=fields[19],process_group=os.getpgrp(),deadline_monotonic=deadline)))
while time.monotonic()<deadline:time.sleep(.002)
(root/'late-write').write_text('UNSAFE publisher survived failure cleanup')
while True:time.sleep(.1)
""".replace("MODE", repr(mode))


def lifecycle_control(root, mode):
    root = Path(root)
    root.mkdir()
    (root / "publisher.py").write_text(publisher_text(mode))
    writer = root / "writer"
    writer.write_text("""#!/usr/bin/python3
import time
with open('stream.fifo','wb',buffering=0) as f:
 f.write(b'x')
 while True:time.sleep(.1)
""")
    writer.chmod(0o700)
    (root / "payload.bin").write_bytes(b"preserve actual unpublished bytes")
    harness = root / "harness.py"
    harness.write_text(
        """import json,pathlib,sys
sys.path.insert(0,SCRIPTS)
import characterize_pcie_clock_trim_stream_v2 as m
root=pathlib.Path(__file__).parent
# Explicit local test substitute: the actual OwnedPublisher lifecycle is used,
# but the child script has no network call and cannot forge release acceptance.
m.publication.__file__=str(root/'publisher.py')
m.previous.PUBLISHER_SHA=m.sha(root/'publisher.py')
def consume(stream,owner):
 assert stream.read(1)==b'x'
 return m.OwnedPublisher(owner)(root/'payload.bin',root/'publication.json')
try:
 m.native_wait(root/'writer',root,root/'native.log',consume)
except BaseException as error:
 (root/'harness-result.json').write_text(json.dumps(dict(status='EXPECTED_FAILURE',error=repr(error))))
 raise SystemExit(23)
raise SystemExit(99)
""".replace("SCRIPTS", repr(str(ROOT / "scripts")))
    )
    process = subprocess.Popen([sys.executable, str(harness)], start_new_session=True)
    armed = None
    start = time.monotonic()
    try:
        if mode != "publisher_nonzero":
            armed = await_file(root / "publisher-armed.json", process)
            actual = m.process_identity(armed["pid"])
            assert actual and actual["start_ticks"] == armed["start_ticks"]
            if mode.startswith("interrupt"):
                # Only parent PID receives SIGINT, never its session or children.
                process.send_signal(signal.SIGINT)
        assert process.wait(timeout=9) == 23
        result = json.loads((root / "harness-result.json").read_text())
        owner = json.loads((root / "owned-processes.json").read_text())
        native = json.loads((root / "native-process.json").read_text())
        assert owner["status"] == "CANCELLED"
        assert owner["elapsed_watchdog_seconds"] is None
        assert len(owner["processes"]) == 2
        assert {x["kind"] for x in owner["processes"]} == {"native", "publisher"}
        assert (
            owner["processes"][0]["process_group"]
            != owner["processes"][1]["process_group"]
        )
        assert all(
            m.process_identity(x["identity"]["pid"]) is None for x in owner["processes"]
        )
        assert native["returncode"] is not None
        observations = stopped(armed) if armed else []
        if armed:
            wait = armed["deadline_monotonic"] + 0.05 - time.monotonic()
            if wait > 0:
                time.sleep(wait)
        assert not (root / "late-write").exists()
        assert (
            root / "payload.bin"
        ).read_bytes() == b"preserve actual unpublished bytes"
        assert not (root / "publication.json").exists()
        if mode.startswith("interrupt"):
            assert "SIGINT" in owner["reason"]
        elif mode == "exited_leader":
            assert "live descendant" in owner["reason"]
        else:
            assert "publication nonzero" in owner["reason"]
        receipt = dict(
            status="PASS_ACTUAL_OWNED_PUBLISHER_FAILURE_CONTROL",
            mode=mode,
            native_executed=False,
            network_calls=False,
            parent_exitcode=process.returncode,
            parent_only_sigint=mode.startswith("interrupt"),
            retained_identity=armed,
            child_nonexecution_observations=observations,
            marker_checked_monotonic=time.monotonic(),
            marker_absent=True,
            elapsed_seconds=time.monotonic() - start,
            lifecycle=owner,
            native_lifecycle=native,
            harness_result=result,
        )
        receipt["outputs"] = {p.name: m.sha(p) for p in root.iterdir() if p.is_file()}
        m.atomic(root / "control.json", receipt)
        return receipt
    finally:
        # Test fails safely even if implementation regresses; never a native watchdog.
        if process.poll() is None:
            os.killpg(process.pid, signal.SIGKILL)
            process.wait()
        if (root / "owned-processes.json").exists():
            for row in json.loads((root / "owned-processes.json").read_text())[
                "processes"
            ]:
                try:
                    os.killpg(row["process_group"], signal.SIGKILL)
                except ProcessLookupError:
                    pass


@pytest.mark.parametrize(
    "mode",
    ["interrupt_direct", "interrupt_descendant", "exited_leader", "publisher_nonzero"],
)
def test_actual_publisher_tree_failure_or_parent_interrupt_cannot_write_late(
    tmp_path, mode
):
    assert lifecycle_control(tmp_path / mode, mode)["status"].startswith("PASS_")


def test_v1_frozen_and_metrology_parser_deck_are_exact_aliases():
    assert m.sha(m.previous.__file__) == m.PREVIOUS_SHA
    assert (
        m.sha(ROOT / "sw/tests/test_pcie_clock_trim_stream_v1.py")
        == m.PREVIOUS_TEST_SHA
    )
    for name in ("Meter", "PartQueue", "long_deck", "replay_local_parts"):
        assert getattr(m, name) is getattr(m.previous, name)


def test_production_run_only_delta_is_owned_publisher_and_consumer_signature():
    def definition(path):
        return next(
            n
            for n in ast.parse(Path(path).read_text()).body
            if isinstance(n, ast.FunctionDef) and n.name == "run"
        )

    old, new = definition(m.previous.__file__), definition(m.__file__)
    old_lambda = next(n for n in ast.walk(old) if isinstance(n, ast.Lambda))
    new_lambda = next(n for n in ast.walk(new) if isinstance(n, ast.Lambda))
    assert ast.dump(new_lambda.args) == ast.dump(
        ast.parse("lambda f, owner: 0", mode="eval").body.args
    )
    assert len(new_lambda.body.keywords) == 1
    assert new_lambda.body.keywords[0].arg == "publisher"
    assert ast.dump(new_lambda.body.keywords[0].value) == ast.dump(
        ast.parse("OwnedPublisher(owner)", mode="eval").body
    )
    new_lambda.args = old_lambda.args
    new_lambda.body.keywords = []
    assert ast.dump(old) == ast.dump(new)


def test_cancel_prevents_any_new_process(tmp_path):
    with m.ProcessOwner(tmp_path / "owner.json") as owner:
        owner.request_cancel("deliberate")
        with pytest.raises(m.Cancelled):
            owner.launch("publisher", [sys.executable, "-c", "raise SystemExit(99)"])
    assert owner.entries == []


def test_cancel_during_spawn_registers_then_reaps_process(tmp_path, monkeypatch):
    original = subprocess.Popen
    with m.ProcessOwner(tmp_path / "owner.json") as owner:

        def spawn(*args, **kwargs):
            process = original(*args, **kwargs)
            owner.request_cancel("actual cancellation during launch registration")
            return process

        monkeypatch.setattr(m.subprocess, "Popen", spawn)
        with pytest.raises(m.Cancelled):
            owner.launch(
                "publisher", [sys.executable, "-c", "import time;time.sleep(20)"]
            )
    assert len(owner.entries) == 1
    assert m.process_identity(owner.entries[0]["process"].pid) is None


def test_early_native_exit_does_not_leave_unscheduled_fifo_reader(tmp_path):
    writer = tmp_path / "writer"
    writer.write_text("#!/usr/bin/python3\nraise SystemExit(17)\n")
    writer.chmod(0o700)
    with pytest.raises(ValueError, match="native nonzero"):
        m.native_wait(
            writer, tmp_path, tmp_path / "run.log", lambda stream, owner: stream.read()
        )
    assert (
        json.loads((tmp_path / "native-process.json").read_text())["returncode"] == 17
    )


def test_healthy_publisher_has_no_elapsed_watchdog(tmp_path):
    with m.ProcessOwner(tmp_path / "owner.json") as owner:
        process = owner.launch(
            "publisher", [sys.executable, "-c", "import time;time.sleep(.25)"]
        )
        assert owner.wait(process) == 0
    receipt = json.loads((tmp_path / "owner.json").read_text())
    assert receipt["status"] == "HEALTHY" and receipt["cleanup"] is None
    assert receipt["elapsed_watchdog_seconds"] is None


def test_evidence_io_failure_cannot_prevent_kill_reap_or_replace_original_error(
    tmp_path, monkeypatch
):
    script = "import pathlib,signal,time;signal.signal(signal.SIGTERM,signal.SIG_IGN);pathlib.Path('armed').write_text('ready');time.sleep(20)"
    owner = m.ProcessOwner(tmp_path / "owner.json")
    with pytest.raises(ValueError, match="original consumer failure"):
        with owner:
            process = owner.launch(
                "publisher", [sys.executable, "-c", script], cwd=tmp_path
            )
            deadline = time.monotonic() + 3
            while not (tmp_path / "armed").exists():
                assert time.monotonic() < deadline
                time.sleep(0.01)

            def full_disk(*args, **kwargs):
                raise OSError(28, "actual injected evidence I/O failure")

            monkeypatch.setattr(m, "atomic", full_disk)
            raise ValueError("original consumer failure")
    assert process.returncode == -signal.SIGKILL
    assert m.process_identity(process.pid) is None
    assert owner.cleanup["recording_errors"]
    assert owner.cleanup["groups"][0]["kill_sent"]


def test_proc_observation_failure_cannot_skip_unconditional_group_kill(
    tmp_path, monkeypatch
):
    owner = m.ProcessOwner(tmp_path / "owner.json")
    with pytest.raises(ValueError, match="original failure"):
        with owner:
            process = owner.launch(
                "publisher", [sys.executable, "-c", "import time;time.sleep(20)"]
            )

            def vanished(group):
                raise ProcessLookupError(3, "injected exact observed proc race")

            monkeypatch.setattr(m, "group_members", vanished)
            raise ValueError("original failure")
    assert process.returncode == -signal.SIGKILL
    assert m.process_identity(process.pid) is None
    assert owner.cleanup["observation_errors"]
    assert owner.cleanup["groups"][0]["kill_sent"]


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    assert not args.out.exists()
    args.out.mkdir()
    cases = [
        lifecycle_control(args.out / mode, mode)
        for mode in (
            "interrupt_direct",
            "interrupt_descendant",
            "exited_leader",
            "publisher_nonzero",
        )
    ]
    result = dict(
        status="PASS_ACTUAL_V2_LIFECYCLE_CONTROLS",
        cases=cases,
        source_sha256={
            str(Path(p).resolve()): m.sha(p)
            for p in (__file__, m.__file__, m.previous.__file__)
        },
        original_aggregate_failure_preserved=True,
        native_spice_executed=False,
    )
    m.atomic(args.out / "result.json", result)
    print(result["status"], m.sha(args.out / "result.json"))


if __name__ == "__main__":
    main()
